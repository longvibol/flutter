2. create product table with Migration 

public function up(): void
    {
        Schema::create('products', function (Blueprint $table) {
            $table->id();
            $table->string('name');
            $table->text('description');
            $table->decimal('price', 8, 2);
            $table->integer('stock');
            $table->string('status')->default('available');
            $table->string('image')->nullable();
            $table->timestamps();
        });
    }


>> php artisan migrate  
>> php artisan migrate:fresh

how to crate fatory 

1. we need model 

>> php artisan make:model Product

<?php

namespace App\Models;

use Illuminate\Database\Eloquent\Model;

class Product extends Model
{
    //
}


- ចឹងវាបានបង្កើត Model

class Product extends Model
{
    protected $fillable = [
        'name',
        'description',
        'price',
        'stock',
        'status',
        'image',
    ];
}

model របស់យើង និង field ដែលយើងត្រូវប្រើ

2. Import Factory to Product model 

<?php

namespace App\Models;

use Illuminate\Database\Eloquent\Model;
use Illuminate\Database\Eloquent\Factories\HasFactory;

class Product extends Model
{
    use HasFactory;
    protected $fillable = [
        'name',
        'description',
        'price',
        'stock',
        'status',
        'image',
    ];
}

we add : use HasFactory; and use Illuminate\Database\Eloquent\Factories\HasFactory;

3. Generate factory command 

>> php artisan make:factory ProductFactory

then go to factories->ProductFactory.php
